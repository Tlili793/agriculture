-- PostgreSQL schema for refrigerator / food inventory application.
-- Tables: Budget, Fruits, Vegetables, Meals

BEGIN;

CREATE TABLE IF NOT EXISTS Budget (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    monthly_limit   NUMERIC(10, 2) NOT NULL CHECK (monthly_limit >= 0),
    month           SMALLINT NOT NULL CHECK (month BETWEEN 1 AND 12),
    year            SMALLINT NOT NULL CHECK (year >= 2000),
    created_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (year, month)
);

CREATE TABLE IF NOT EXISTS Fruits (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name            VARCHAR(100) NOT NULL,
    quantity        NUMERIC(10, 2) NOT NULL DEFAULT 0 CHECK (quantity >= 0),
    unit            VARCHAR(20) NOT NULL DEFAULT 'piece',
    price           NUMERIC(10, 2) NOT NULL DEFAULT 0 CHECK (price >= 0),
    purchase_date   DATE,
    expiration_date DATE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CHECK (
        expiration_date IS NULL
        OR purchase_date IS NULL
        OR expiration_date >= purchase_date
    )
);

CREATE TABLE IF NOT EXISTS Vegetables (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name            VARCHAR(100) NOT NULL,
    quantity        NUMERIC(10, 2) NOT NULL DEFAULT 0 CHECK (quantity >= 0),
    unit            VARCHAR(20) NOT NULL DEFAULT 'piece',
    price           NUMERIC(10, 2) NOT NULL DEFAULT 0 CHECK (price >= 0),
    purchase_date   DATE,
    expiration_date DATE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CHECK (
        expiration_date IS NULL
        OR purchase_date IS NULL
        OR expiration_date >= purchase_date
    )
);

CREATE TABLE IF NOT EXISTS Meals (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name            VARCHAR(100) NOT NULL,
    description     TEXT,
    meal_type       VARCHAR(20) NOT NULL
                    CHECK (meal_type IN ('BREAKFAST', 'LUNCH', 'DINNER', 'SNACK')),
    planned_date    DATE,
    estimated_cost  NUMERIC(10, 2) NOT NULL DEFAULT 0
                    CHECK (estimated_cost >= 0),
    instructions    TEXT,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Useful indexes
CREATE INDEX IF NOT EXISTS idx_fruits_expiration
    ON Fruits (expiration_date);

CREATE INDEX IF NOT EXISTS idx_vegetables_expiration
    ON Vegetables (expiration_date);

CREATE INDEX IF NOT EXISTS idx_meals_planned_date
    ON Meals (planned_date);

COMMIT;
